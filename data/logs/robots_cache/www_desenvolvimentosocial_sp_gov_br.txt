    function extrairPartesDominio(url) {
      if (!url || typeof url !== 'string') return [];
      let dominio = url.toLowerCase()
        .replace(/^https?:\/\//, '')   
        .replace(/^www\./i, '');      
      const indiceCorte = dominio.search(/[/?&#:]/);
      if (indiceCorte !== -1) {
        dominio = dominio.substring(0, indiceCorte);
      }
      dominio = dominio.replace(/\.sp\.gov\.br$/i, '');
      return dominio
        .trim()
        .split('.')
        .filter(part => part.length > 0);
    }

 function temPalavraEmComum(array1 = [], array2 = []) {
  const PALAVRAS_CHAVE = new Set(['Poupatempo'].map(s => s.toLowerCase()));

  return [...array1, ...array2].some(item => {
    if (typeof item !== 'string') return false;
    return PALAVRAS_CHAVE.has(item.trim().toLowerCase());
  });
}

function initExternalLinkHandler() {
  const links = document.querySelectorAll('ul.navbar-nav.mr-auto a[href]');
  
  links.forEach(link => {
    link.addEventListener('click', function(e) {
      const href = this.getAttribute('href');
      
      if (!href || href.startsWith('#') || href.startsWith('javascript:')) {
        return;
      }
      
      const currentDomain = window.location.hostname;
      const tempLink = document.createElement('a');
      tempLink.href = href;
      const linkDomain = tempLink.hostname;
       
      const currentDomainParts = extrairPartesDominio(currentDomain); 
       const linkDomainParts = extrairPartesDominio(linkDomain);

          if (temPalavraEmComum(currentDomainParts, linkDomainParts)) {
            link.target = '_self';  
            return;
          }
      
      if (linkDomain && linkDomain !== currentDomain) {
        e.preventDefault();
        window.open(href, '_blank', 'noopener,noreferrer');
      }
    });
  });
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', initExternalLinkHandler);
} else {
  initExternalLinkHandler();
}
